        function syncSubtitlesFromText() {
            const rawText = document.getElementById('').value.trim();
            if (!rawText) { editorWords = []; renderTimelineTracks(); return; }
            const rawWords = rawText.split(/\s+/);
            const wordDur = (currentClip.end - currentClip.start) / rawWords.length;
            const newEditorWords = [];
            rawWords.forEach((txt, i) => {
                if (editorWords[i]) {
                    newEditorWords.push({...editorWords[i], text: txt});
                } else {
                    newEditorWords.push({id: i, text: txt, start: currentClip.start + (i * wordDur), end: currentClip.start + ((i + 1) * wordDur)});
                }
            });
            editorWords = newEditorWords;
            renderTimelineTracks();
            saveCurrentClipState();
        }
        
        async function saveCurrentClipState() {

                });
            } catch (err) { console.error("", err); }
        }
        
        let draggingWord = null, draggingHandle = null, dragStartX = 0, dragStartVal = 0;
        function startWordDrag(e, word, handle) {
            e.stopPropagation(); e.preventDefault();
            draggingWord = word; draggingHandle = handle; dragStartX = e.clientX;
            dragStartVal = handle === '' ? word.start : word.end;
            document.addEventListener('', onWordDrag);
            document.addEventListener('', onWordDragEnd);
        }
        function onWordDrag(e) {
            if(!draggingWord) return;
            const subsTrack = document.getElementById('');
            const totalDur = currentClip.end - currentClip.start;
            const trackWidth = subsTrack.getBoundingClientRect().width;
            const dx = e.clientX - dragStartX;
            const dt = (dx / trackWidth) * totalDur;
            if (draggingHandle === '') draggingWord.start = Math.max(currentClip.start, Math.min(draggingWord.end - 0.1, dragStartVal + dt));
            else draggingWord.end = Math.max(draggingWord.start + 0.1, Math.min(currentClip.end, dragStartVal + dt));
            renderTimelineTracks();
        }
        function onWordDragEnd() {
            if(draggingWord) saveCurrentClipState();
            draggingWord = null;
            document.removeEventListener('', onWordDrag);
            document.removeEventListener('', onWordDragEnd);
        }

        function uploadSticker(e) {
            const file = e.target.files[0];
            if(!file) return;
            const reader = new FileReader();
            reader.onload = (ev) => {
                const id = Date.now().toString();
                editorStickers.push({
                    id: id,
                    name: file.name,
                    file: file,
                    dataUrl: ev.target.result,
                    x: 100, y: 100, w: 200, h: 200
                });
                renderStickersList();
                renderStickersCanvas();
            };
            reader.readAsDataURL(file);
        }

        function renderStickersCanvas() {
            
            document.querySelectorAll('').forEach(el => el.remove());
            const workspace = document.getElementById('');
            
            editorStickers.forEach(s => {
                const div = document.createElement('');
                div.className = '';
                div.id = '' + s.id;
                div.style.position = '';
                div.style.border = '';
                div.style.background = '';
                div.style.cursor = '';
                div.style.zIndex = '';
                
                
                const scaleX = workspace.clientWidth / VID_W;
                const scaleY = workspace.clientHeight / VID_H;
                
                div.style.left = (s.x * scaleX) + '';
                div.style.top = (s.y * scaleY) + '';
                div.style.width = (s.w * scaleX) + '';
                div.style.height = (s.h * scaleY) + '';
                
                div.innerHTML = ``;
                
                workspace.appendChild(div);
                makeDraggable(div, (rect) => {
                    s.x = rect.x / scaleX;
                    s.y = rect.y / scaleY;
                    s.w = rect.w / scaleX;
                    s.h = rect.h / scaleY;
                });
            });
        }
